let currentExList = []; let currentIndex = 0; let timeLeft = 0; let isWorking = false; let timerId = null; let wakeLock = null; // New variable to hold the screen lock const chime = new Audio('https://actions.google.com/sounds/v1/alarms/beep_short.ogg'); // NEW: Function to keep screen alive async function requestWakeLock() { try { if ('wakeLock' in navigator) { wakeLock = await navigator.wakeLock.request('screen'); } } catch (err) { console.log(`${err.name}, ${err.message}`); } } // NEW: Function to allow screen to sleep again function releaseWakeLock() { if (wakeLock !== null) { wakeLock.release(); wakeLock = null; } } window.onload = function() { const list = document.getElementById('session-list'); list.innerHTML = ""; sessions.forEach((s, idx) => { const div = document.createElement('div'); div.className = 'menu-item'; div.onclick = () => selectSession(idx); div.innerHTML = `

${s.title}

${s.exercises.length} Movements • Tap to Start

`; list.appendChild(div); }); }; function selectSession(idx) { currentExList = sessions[idx].exercises; currentIndex = 0; setupExercise(); document.getElementById('menu-screen').classList.add('hidden'); document.getElementById('timer-screen').classList.remove('hidden'); } function setupExercise() { isWorking = false; timeLeft = currentExList[currentIndex].prep; updateUI(); } function skipExercise() { if (currentIndex < currentExList.length - 1) { currentIndex++; setupExercise(); } else { finish(); } } function prevExercise() { if (currentIndex > 0) { currentIndex--; setupExercise(); } } function toggleTimer() { if (timerId) { clearInterval(timerId); timerId = null; document.getElementById('startBtn').innerText = "RESUME"; releaseWakeLock(); // Let phone sleep if paused } else { document.getElementById('startBtn').innerText = "PAUSE"; timerId = setInterval(tick, 1000); requestWakeLock(); // Keep screen on while running } } function tick() { if (timeLeft > 0) { timeLeft--; } else { try { chime.play(); } catch(e) {} if (!isWorking) { isWorking = true; timeLeft = currentExList[currentIndex].work; } else { skipExercise(); return; } } updateUI(); } function updateUI() { const ex = currentExList[currentIndex]; document.getElementById('ex-name').innerText = ex.name; document.getElementById('ex-desc').innerText = ex.description; document.getElementById('phase').innerText = isWorking ? "EXERCISE" : "PREP"; document.getElementById('phase').style.color = isWorking ? "var(--accent)" : "var(--prep)"; document.getElementById('display').innerText = `00:${timeLeft.toString().padStart(2, '0')}`; document.getElementById('progress').innerText = `Step ${currentIndex + 1} of ${currentExList.length}`; } function finish() { if (timerId) clearInterval(timerId); timerId = null; releaseWakeLock(); // Session over, let phone sleep document.getElementById('phase').innerText = "FINISHED"; document.getElementById('ex-name').innerText = "Session Complete"; document.getElementById('ex-desc').innerText = "Well done!"; document.getElementById('display').innerText = "00:00"; document.getElementById('startBtn').classList.add('hidden'); document.getElementById('navRow').classList.add('hidden'); }
